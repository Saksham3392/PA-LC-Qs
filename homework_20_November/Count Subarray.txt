import java.util.*;

class Solution {
    public int subarraySum(int[] nums, int k) {
        Map<Integer, Integer> frequency = new HashMap<>();
        frequency.put(0, 1);
        int prefix = 0;
        int count = 0;
        for (int value : nums) {
            prefix += value;
            count += frequency.getOrDefault(prefix - k, 0);
            frequency.put(prefix, frequency.getOrDefault(prefix, 0) + 1);
        }
        return count;
    }
}
