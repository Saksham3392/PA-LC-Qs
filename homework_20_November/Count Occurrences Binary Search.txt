class Solution {
    public int countOccurrences(int[] nums, int target) {
        int first = bound(nums, target, false);
        if (first == nums.length || nums[first] != target) return 0;
        int afterLast = bound(nums, target, true);
        return afterLast - first;
    }

    private int bound(int[] nums, int target, boolean upper) {
        int left = 0;
        int right = nums.length;
        while (left < right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] < target || (upper && nums[mid] == target)) left = mid + 1;
            else right = mid;
        }
        return left;
    }
}
