class Solution {
    static class Node {
        int data;
        Node prev;
        Node next;

        Node(int data) {
            this.data = data;
        }
    }

    public boolean isPalindrome(Node head) {
        if (head == null) return true;
        Node right = head;
        while (right.next != null) right = right.next;
        Node left = head;
        while (left != right && left.prev != right) {
            if (left.data != right.data) return false;
            left = left.next;
            right = right.prev;
        }
        return true;
    }
}
