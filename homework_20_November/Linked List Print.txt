import java.util.*;

class Solution {
    static class Node {
        int data;
        Node next;

        Node(int data) {
            this.data = data;
        }
    }

    public List<Integer> printList(Node head) {
        List<Integer> values = new ArrayList<>();
        for (Node current = head; current != null; current = current.next) values.add(current.data);
        return values;
    }
}
