import java.util.*;

class Solution {
    public List<Integer> findDuplicates(int[] nums) {
        List<Integer> result = new ArrayList<>();
        for (int value : nums) {
            int index = Math.abs(value) - 1;
            if (nums[index] < 0) result.add(index + 1);
            else nums[index] = -nums[index];
        }
        return result;
    }
}
