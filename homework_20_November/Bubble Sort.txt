class Solution {
    public int[] bubbleSort(int[] nums) {
        int[] result = nums.clone();
        for (int i = 0; i < result.length - 1; i++) {
            boolean swapped = false;
            for (int j = 0; j < result.length - i - 1; j++) {
                if (result[j] > result[j + 1]) {
                    int temp = result[j];
                    result[j] = result[j + 1];
                    result[j + 1] = temp;
                    swapped = true;
                }
            }
            if (!swapped) break;
        }
        return result;
    }
}
