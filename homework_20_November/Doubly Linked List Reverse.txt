class Solution {
    static class Node {
        int data;
        Node prev;
        Node next;

        Node(int data) {
            this.data = data;
        }
    }

    public Node reverse(Node head) {
        Node current = head;
        Node newHead = null;
        while (current != null) {
            Node next = current.next;
            current.next = current.prev;
            current.prev = next;
            newHead = current;
            current = next;
        }
        return newHead;
    }
}
